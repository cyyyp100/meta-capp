# Meta-Capp — Python dependencies.
# Versions aligned with the working conda env `nwol`.
#
# Quick start:
#   conda create -n nwol python=3.11.9 && conda activate nwol
#   pip install -r requirements.txt
#   cd frontend && npm install && npm run build && cd ..   # web UI bundle
#   python main.py                                          # native app window
#
# Prerequisites NOT installable via pip:
#   - Node.js + npm  (to build the React/Vite frontend in frontend/)
#   - Ollama running locally with the model from config/settings.py
#     (OLLAMA_MODEL, default `gemma4:e4b`):  ollama pull gemma4:e4b
#     The app degrades gracefully when Ollama is offline (no API key needed).
