# Meta-Capp — Python dependencies.
# Versions aligned with the working conda env `metacapp`.
#
# Quick start:
#   conda create -n metacapp python=3.11.9 && conda activate metacapp
#   pip install -r requirements.txt
#   cd frontend && npm install && npm run build && cd ..   # web UI bundle
#   python main.py                                          # native app window
#
# Prerequisites NOT installable via pip:
#   - Node.js + npm  (to build the React/Vite frontend in frontend/)
#   - Ollama running locally with one model of the gemma4 family
#     (config/settings.py:LLM_MODELS; which one is a setting, « auto » by
#     default, and Settings ▸ Clikoda shows the `ollama pull` command):
#       ollama pull gemma4:e4b    # 9.6 GB, needs 12 GB of RAM — the standard one
#       ollama pull gemma4:e2b    # 7.2 GB, needs 8 GB — for a lighter machine
#       ollama pull qwen3:1.7b    # 1.4 GB, needs 4 GB — the tiny tier, economy mode
#     GPU optional. The app degrades gracefully when Ollama is offline (no API
#     key needed).
#   - Optional: the embedding model (OLLAMA_EMBED_MODEL, `embeddinggemma`, ~600 MB)
#     for semantic search inside the document being read:  ollama pull embeddinggemma
#     Without it the reader's search stays lexical (BM25) — nothing breaks.

# PDF rendering, page text and chapter detection.
# pypdfium2 wraps PDFium (the PDF engine of Chrome), licensed Apache-2.0 OR
# BSD-3-Clause — deliberately NOT PyMuPDF, which is AGPL-3.0 and would force
# the whole app open-source or a commercial Artifex licence.
# PDFium parses every PDF the user imports: it is the widest attack surface of
# the app, so the pin follows PDFium's security releases (5.14.0 = PDFium
# 156.0.8076; 4.30.0 shipped the June 2024 engine). The whole suite passes on
# both. OCR tools that pin pypdfium2==4.30.0 (marker-pdf, surya) belong in
# their own environment, not in Meta-Capp's.
# PDFium is NOT thread-safe — every call is serialised through
# pdf_viewer/engine.py:PDFIUM_LOCK. Do not bypass it.
pypdfium2==5.14.0
Pillow==13.0.0            # PNG encoding of the rendered pages (PdfBitmap -> file)

# Local API server (the app's backend) — runs on 127.0.0.1, embedded in the app
fastapi==0.136.1
uvicorn==0.46.0
websockets==16.0          # WebSocket transport (reader + brainstorming streams)
pydantic==2.10.6          # request bodies in server/routers (used directly)

# Native desktop shell (embeds the local web UI in an OS window)
pywebview==6.2.1
# Windows-only: pywebview's EdgeChromium/WebView2 backend needs the .NET binding.
pythonnet==3.0.5 ; sys_platform == "win32"
# Linux: the native window needs GTK or Qt, deliberately NOT pinned here —
# GTK requires system libraries pip cannot install, Qt is ~200 MB. Without
# either, the app opens in the default browser (desktop/pywebview_main.py).
# Pick one:  pip install "pywebview[gtk]"   or   pip install "pywebview[qt]"
# (system packages for GTK: README.md, "Linux: getting a native window").

# Pinyin of the Mandarin lessons (services/lang_mandarin.py). MIT code, pure
# Python, no network; dictionaries derived from Unihan, CC-CEDICT and zdic.
# Imported optionally: without it Mandarin stays readable but has no pinyin.
# Ships its own PyInstaller hook, which collects its JSON dictionaries.
pypinyin==0.55.0

# HTTP client — required by FastAPI's TestClient
httpx==0.28.1

# Mozilla CA bundle for the opt-in update check (services/updates.py). Frozen
# builds cannot rely on OpenSSL's compiled-in store: it points at the build
# machine. Already pulled in by httpx; pinned here because the app imports it.
certifi==2026.10.19

# Testing (reportlab builds the PDF fixtures — test-only, never shipped)
pytest==9.0.3
reportlab==5.0.1
