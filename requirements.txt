# Meta-Capp — Python dependencies.
# Versions aligned with the working conda env `nwol`.
#
# Quick start:
#   conda create -n nwol python=3.11.9 && conda activate nwol
#   pip install -r requirements.txt
#   cd frontend && npm install && npm run build && cd ..   # web UI bundle
#   python main.py                                          # native app window
#
# Prerequisites NOT installable via pip:
#   - Node.js + npm  (to build the React/Vite frontend in frontend/)
#   - Ollama running locally with the model from config/settings.py
#     (OLLAMA_MODEL, default `gemma4:e4b`):  ollama pull gemma4:e4b
#     The app degrades gracefully when Ollama is offline (no API key needed).
#   - Optional: the embedding model (OLLAMA_EMBED_MODEL, `embeddinggemma`, ~600 MB)
#     for semantic search inside the document being read:  ollama pull embeddinggemma
#     Without it the reader's search stays lexical (BM25) — nothing breaks.

# PDF rendering, page text and chapter detection.
# pypdfium2 wraps PDFium (the PDF engine of Chrome), licensed Apache-2.0 OR
# BSD-3-Clause — deliberately NOT PyMuPDF, which is AGPL-3.0 and would force
# the whole app open-source or a commercial Artifex licence.
# See architecture/07-pdf-et-code-reader.md.
# Pinned to 4.30.0, not the latest 5.x: that is the version the wider Python PDF
# ecosystem pins, and upgrading breaks other tools sharing the `nwol` env.
# PDFium is NOT thread-safe — every call is serialised through
# pdf_viewer/engine.py:PDFIUM_LOCK. Do not bypass it.
pypdfium2==4.30.0
Pillow==12.3.0            # PNG encoding of the rendered pages (PdfBitmap -> file)

# Local API server (the app's backend) — runs on 127.0.0.1, embedded in the app
fastapi==0.136.1
uvicorn==0.46.0
websockets==16.0          # WebSocket transport (reader + brainstorming streams)
pydantic==2.10.6          # request bodies in server/routers (used directly)

# Native desktop shell (embeds the local web UI in an OS window)
pywebview==6.2.1
# Windows-only: pywebview's EdgeChromium/WebView2 backend needs the .NET binding.
pythonnet==3.0.5 ; sys_platform == "win32"

# HTTP client — required by FastAPI's TestClient
httpx==0.28.1

# Mozilla CA bundle for the opt-in update check (services/updates.py). Frozen
# builds cannot rely on OpenSSL's compiled-in store: it points at the build
# machine. Already pulled in by httpx; pinned here because the app imports it.
certifi==2026.1.4

# Testing (reportlab builds the PDF fixtures — test-only, never shipped)
pytest==9.0.3
reportlab==4.4.10
